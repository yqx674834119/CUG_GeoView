1. Check pod status:
   kubectl get pods -n {{ .Release.Namespace }}

2. Check services:
   kubectl get svc -n {{ .Release.Namespace }}

3. Check app logs:
   kubectl logs deploy/{{ include "geoview.appName" . }} -n {{ .Release.Namespace }}

4. If mysql is enabled:
   kubectl logs sts/{{ include "geoview.mysqlName" . }} -n {{ .Release.Namespace }}
